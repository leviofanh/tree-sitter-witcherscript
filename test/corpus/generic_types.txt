function LoadPreset( optional presetMap : map<int, float> ) : map<int, float>
{
	var i : int;
	for (i = 0; i < presetMap.Size(); i += 1)
	{
		presetMap[presetMap.Key(i)] = 0.f;
	}
	return presetMap;
}
